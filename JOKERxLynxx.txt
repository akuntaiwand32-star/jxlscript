local themeGrads, themeStrokes, themeScrolls, themeTexts = {}, {}, {}, {}

local function regText(o)
local role
if o.TextColor3 == C.text then role = "text"
elseif o.TextColor3 == C.muted then role = "muted"
elseif o.TextColor3 == C.pink then role = "accent" end
if role then table.insert(themeTexts, {o, role}) end
end
local function stroke(o, color, th, tr)
local st = make("UIStroke", {Color = color, Thickness = th or 1, Transparency = tr or 0,
ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, o)
if color == C.purple then table.insert(themeStrokes, st) end
return st
end
local function gradient(o, c1, c2, rot)
local g = make("UIGradient", {Color = ColorSequence.new(c1, c2), Rotation = rot or 0}, o)
if c1 == C.purple and c2 == C.pink then table.insert(themeGrads, g) end
return g
end

local function place()

local function isEggTool0(t)
if not t:IsA("Tool") then return false end
if t:GetAttribute("IsEggTool") or t:GetAttribute("EggId") ~= nil then return true end
local n = t.Name:lower()
return n:find("egg", 1, true) ~= nil or n:find("telur", 1, true) ~= nil
end
local function carrying()
local ch, bp = player.Character, player:FindFirstChildOfClass("Backpack")
for _, c in ipairs({ch, bp}) do
if c then for _, t in ipairs(c:GetChildren()) do if isEggTool0(t) then return true end end end
end
return false
end
local function visibleGui(o)
local g = o
while g and g:IsA("GuiObject") do
if not g.Visible then return false end
g = g.Parent
end
local sg = o:FindFirstAncestorOfClass("ScreenGui")
return not (sg and not sg.Enabled)
end
local function findDrop()
local pg = player:FindFirstChildOfClass("PlayerGui")
if not pg then return nil end
local label
for _, d in ipairs(pg:GetDescendants()) do
if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.Text ~= "" and visibleGui(d) and d.AbsoluteSize.X > 0 then
local t = d.Text:lower()
if t:find("drop", 1, true) or t:find("jatuhkan", 1, true) then label = d break end
end
end
if not label then return nil end
local c = label.AbsolutePosition + label.AbsoluteSize / 2
local best, bestArea
for _, b in ipairs(pg:GetDescendants()) do
if b:IsA("GuiButton") and visibleGui(b) then
local p, sz = b.AbsolutePosition, b.AbsoluteSize
if c.X >= p.X and c.X <= p.X + sz.X and c.Y >= p.Y and c.Y <= p.Y + sz.Y then
local area = sz.X * sz.Y
if area > 0 and (not bestArea or area < bestArea) then best, bestArea = b, area end
end
end
end
return best, label
end
local function press(btn, label, useClick)
if btn and not useClick then
local fired = false
if getconnections then
for _, sig in ipairs({"Activated", "MouseButton1Click"}) do
if not fired then
pcall(function()
for _, cn in ipairs(getconnections(btn[sig])) do
fired = true
pcall(function() cn:Fire() end)
end
end)
end
end
end
if not fired and firesignal then pcall(function() firesignal(btn.Activated) end) end
elseif label then
pcall(function()
local vim = game:GetService("VirtualInputManager")
local inset = game:GetService("GuiService"):GetGuiInset()
local p = label.AbsolutePosition + label.AbsoluteSize / 2 + inset
vim:SendMouseButtonEvent(p.X, p.Y, 0, true, game, 0)
task.wait(0.05)
vim:SendMouseButtonEvent(p.X, p.Y, 0, false, game, 0)
end)
end
end
local function unstick()
local c = player.Character
local h = c and c:FindFirstChildOfClass("Humanoid")
local r = c and c:FindFirstChild("HumanoidRootPart")
if r then
r.Anchored = false
r.AssemblyLinearVelocity = Vector3.zero
end
if h then
h.PlatformStand = false
h.Sit = false
pcall(function() h:ChangeState(Enum.HumanoidStateType.GettingUp) end)
end
if c then
for _, part in ipairs(c:GetChildren()) do
if part:IsA("BasePart") and part.Anchored then part.Anchored = false end
end
end
end
local function dropEgg()
for try = 1, 3 do
local btn, label = findDrop()
if label then press(btn, label, try == 2 or not btn) end
task.wait(1.2)
unstick()
if not carrying() then return true end
end
return false
end
