local function regText(o)
local role
if o.TextColor3 == C.text then role = "text"
elseif o.TextColor3 == C.muted then role = "muted"
elseif o.TextColor3 == C.pink then role = "accent" end
if role then table.insert(themeTexts, {o, role}) end
end
local function stroke(o, color, th, tr)
local st = make("UIStroke", {Color = color, Thickness = th or 1, Transparency = tr or 0,
ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, o)
if color == C.purple then table.insert(themeStrokes, st) end
return st
end
local function gradient(o, c1, c2, rot)
local g = make("UIGradient", {Color = ColorSequence.new(c1, c2), Rotation = rot or 0}, o)
if c1 == C.purple and c2 == C.pink then table.insert(themeGrads, g) end
return g
end

local lagOn = false
local saved = setmetatable({}, {__mode = "k"})

local qualOrig
local function setProps(inst, changes)
local s = saved[inst]
if not s then
s = {}
saved[inst] = s
end
for k, v in pairs(changes) do
pcall(function()
if s[k] == nil then s[k] = inst[k] end
inst[k] = v
end)
end
end
local function boost(o)
if o:IsA("Terrain") then return end
if o:IsA("BasePart") then
setProps(o, {Material = Enum.Material.SmoothPlastic, Reflectance = 0, CastShadow = false})
elseif o:IsA("Texture") or (o:IsA("Decal") and not (o.Parent and o.Parent.Name == "Head")) then
setProps(o, {Transparency = 1})
elseif o:IsA("ParticleEmitter") then
setProps(o, {Enabled = false, Rate = 0, Lifetime = NumberRange.new(0)})
elseif o:IsA("Trail") or o:IsA("Beam") or o:IsA("Smoke") or o:IsA("Fire")
or o:IsA("Sparkles") or o:IsA("PostEffect") or o:IsA("Clouds") then
setProps(o, {Enabled = false})
elseif o:IsA("ForceField") then
setProps(o, {Visible = false})
elseif o:IsA("Atmosphere") then
setProps(o, {Density = 0, Offset = 0, Glare = 0, Haze = 0})
end
end
local function applyWorld()
setProps(Lighting, {
GlobalShadows = false, FogEnd = 9e9,
EnvironmentDiffuseScale = 0, EnvironmentSpecularScale = 0,
})
if Terrain then
setProps(Terrain, {WaterWaveSize = 0, WaterWaveSpeed = 0, WaterReflectance = 0, Decoration = false})
end
pcall(function()
qualOrig = qualOrig or settings().Rendering.QualityLevel
settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
end)
end
local lagJob = 0
local function setLag(v)
lagOn = v
lagJob = lagJob + 1
local my = lagJob
if v then
applyWorld()
task.spawn(function()
local n = 0
for _, o in ipairs(game:GetDescendants()) do
if not lagOn or lagJob ~= my then break end
pcall(boost, o)
n = n + 1
if n % 400 == 0 then task.wait() end
end
end)
else
local list = {}
for inst, props in pairs(saved) do list[inst] = props end
for inst, props in pairs(list) do
for k, val in pairs(props) do
pcall(function() inst[k] = val end)
end
saved[inst] = nil
end
if qualOrig then
pcall(function() settings().Rendering.QualityLevel = qualOrig end)
qualOrig = nil
end
end
end

local function start(p)
stop()
holdMsg = nil
target = p
if activeTween then activeTween:Cancel() activeTween = nil end
local t0, hitFrames, lastNc, lastInfo = os.clock(), 0, 0, 0
local lastPos, lastT, lastSeen, now0
local function step(skipDetect)
local tp = target
if not tp or not tp.Parent then stop("Pemain keluar, berhenti") if rebuild then rebuild() end return end
local ch = tp.Character
local tr = ch and (ch:FindFirstChild("HumanoidRootPart") or ch.PrimaryPart or ch:FindFirstChildWhichIsA("BasePart"))
local hrp = getHRP()
if tr then lastSeen = tr.CFrame
elseif hrp and ch and not tr then
local okp, pv = pcall(function() return ch:GetPivot() end)
local goal = (okp and pv and pv.Position.Magnitude > 1) and pv or lastSeen
if goal then
pcall(function() player:RequestStreamAroundAsync(goal.Position) end)
if now0 == nil or os.clock() - now0 > 0.3 then
now0 = os.clock()
hrp.CFrame = goal * CFrame.new(0, 0, dists[dIdx])
hrp.AssemblyLinearVelocity = Vector3.zero
end
end
end
if not tr or not hrp then
if os.clock() - lastInfo > 0.5 then lastInfo = os.clock() sw.sub.Text = "Menunggu karakter " .. tp.DisplayName end
lastPos = nil
return
end
local now = os.clock()
if autoStop and not skipDetect and lastPos and lastT and now - t0 > 1.5 then
local dt = math.max(now - lastT, 1e-3)
local sp = (tr.Position - lastPos).Magnitude / dt

local function visibleGui(o)

local function findDrop()
local pg = player:FindFirstChildOfClass("PlayerGui")
if not pg then return nil end
local label
for _, d in ipairs(pg:GetDescendants()) do
if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.Text ~= "" and visibleGui(d) and d.AbsoluteSize.X > 0 then
local t = d.Text:lower()
if t:find("drop", 1, true) or t:find("jatuhkan", 1, true) then label = d break end
end
end
if not label then return nil end
local c = label.AbsolutePosition + label.AbsoluteSize / 2
local best, bestArea
for _, b in ipairs(pg:GetDescendants()) do
if b:IsA("GuiButton") and visibleGui(b) then
local p, sz = b.AbsolutePosition, b.AbsoluteSize
if c.X >= p.X and c.X <= p.X + sz.X and c.Y >= p.Y and c.Y <= p.Y + sz.Y then
local area = sz.X * sz.Y
if area > 0 and (not bestArea or area < bestArea) then best, bestArea = b, area end
end
end
end
return best, label
end

local function r3(n) return tostring(math.floor(n * 1000 + 0.5) / 1000) end
local function fmt(v)
local t = typeof(v)
if t == "number" then return r3(v)
elseif t == "string" then return v
elseif t == "Vector3" then return r3(v.X) .. ", " .. r3(v.Y) .. ", " .. r3(v.Z)
elseif t == "Vector2" then return r3(v.X) .. ", " .. r3(v.Y)
elseif t == "Color3" then return math.floor(v.R * 255 + 0.5) .. ", " .. math.floor(v.G * 255 + 0.5) .. ", " .. math.floor(v.B * 255 + 0.5)
elseif t == "UDim2" then return r3(v.X.Scale) .. ", " .. v.X.Offset .. ", " .. r3(v.Y.Scale) .. ", " .. v.Y.Offset
elseif t == "UDim" then return r3(v.Scale) .. ", " .. v.Offset
elseif t == "CFrame" then local p = v.Position return r3(p.X) .. ", " .. r3(p.Y) .. ", " .. r3(p.Z)
elseif t == "BrickColor" or t == "EnumItem" or t == "Instance" then return v.Name
elseif t == "nil" then return "nil" end
return tostring(v)
end
local function nums(s)
local r = {}
for n in s:gmatch("-?%d+%.?%d*") do table.insert(r, tonumber(n)) end
return r
end
local function parse(old, text)
local t = typeof(old)
if t == "number" then return tonumber(text)
elseif t == "string" then return text end
local n = nums(text)
if t == "Vector3" and #n >= 3 then return Vector3.new(n[1], n[2], n[3])
elseif t == "Vector2" and #n >= 2 then return Vector2.new(n[1], n[2])
elseif t == "Color3" and #n >= 3 then
if n[1] <= 1 and n[2] <= 1 and n[3] <= 1 and text:find("%.") then return Color3.new(n[1], n[2], n[3]) end
return Color3.fromRGB(math.clamp(n[1], 0, 255), math.clamp(n[2], 0, 255), math.clamp(n[3], 0, 255))
elseif t == "UDim2" and #n >= 4 then return UDim2.new(n[1], n[2], n[3], n[4])
elseif t == "UDim" and #n >= 2 then return UDim.new(n[1], n[2])
elseif t == "CFrame" and #n >= 3 then return CFrame.new(n[1], n[2], n[3]) * (old - old.Position)
elseif t == "BrickColor" then
local ok, b = pcall(BrickColor.new, text)
return ok and b or nil
end
return nil
end
local function pathOf(inst)
local parts, o = {}, inst
while o and o ~= game do table.insert(parts, 1, o) o = o.Parent end
if #parts == 0 then return "game" end
local s = 'game:GetService("' .. parts[1].ClassName .. '")'
if parts[1].ClassName ~= parts[1].Name then s = 'game:FindFirstChild("' .. parts[1].Name .. '")' end
for i = 2, #parts do
local nm = parts[i].Name
if nm:match("^[%a_][%w_]*$") then s = s .. "." .. nm else s = s .. '["' .. nm .. '"]' end
end
return s
end

local t = grid[k]

local function nearItem(pos)

for dx = -span, span do

if t then
for _, ip in ipairs(t) do
local d = ip - pos
if d.X * d.X + d.Z * d.Z <= r2 then return true end
end
end
end
end

local function applyTextTheme(c1, c2)
local oldText, oldMuted, oldAccent = C.text, C.muted, C.accent
C.text = baseText:Lerp(c1, 0.10)
C.muted = baseMuted:Lerp(c1, 0.28)
C.accent = c2:Lerp(Color3.new(1, 1, 1), 0.2)
for i = #themeTexts, 1, -1 do
local e = themeTexts[i]
local o, role = e[1], e[2]
if not o.Parent then
table.remove(themeTexts, i)
else
local old, new = oldAccent, C.accent
if role == "text" then old, new = oldText, C.text
elseif role == "muted" then old, new = oldMuted, C.muted end
if o.TextColor3 == old then o.TextColor3 = new end
end
end
end

local function isCosmetic(o)
return o:IsA("Accessory") or o:IsA("Shirt") or o:IsA("Pants") or o:IsA("ShirtGraphic")
or o:IsA("CharacterMesh") or o:IsA("BodyColors")
end
local function attachAccessory(ch, hum, src)
local c = src:Clone()
local h = c:FindFirstChild("Handle")
if not h then return false end
for _, w in ipairs(h:GetChildren()) do
if w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D") then w:Destroy() end
end
pcall(function() hum:AddAccessory(c) end)
if not c.Parent then c.Parent = ch end
task.wait()
local function welded()
for _, w in ipairs(h:GetChildren()) do
if (w:IsA("Weld") or w:IsA("Motor6D")) and w.Part1 and w.Part1:IsDescendantOf(ch) then return true end
end
return false
end
if welded() then return true end
local att = h:FindFirstChildOfClass("Attachment")
if not att then return false end
local target
for _, d in ipairs(ch:GetDescendants()) do
if d:IsA("Attachment") and d.Name == att.Name and d.Parent ~= h and not d:IsDescendantOf(c) then
target = d
break
end
end
if not target then return false end
local weld = Instance.new("Weld")
weld.Name = "AccessoryWeld"
weld.Part0 = h
weld.Part1 = target.Parent
weld.C0 = att.CFrame
weld.C1 = target.CFrame
weld.Parent = h
return true
end

local function applyModel(src)

for _, o in ipairs(ch:GetChildren()) do
if isCosmetic(o) then o:Destroy() end

if o:IsA("Accessory") then

local ok, res = pcall(attachAccessory, ch, hum, o)
if ok and res then accOk = accOk + 1
elseif not ok then warn("[JxL] aksesoris gagal: " .. tostring(res)) end
elseif isCosmetic(o) then
o:Clone().Parent = ch
end
end

end

end
local function snapshot()
local ch = player.Character
if not ch then return end
local m = Instance.new("Model")
for _, o in ipairs(ch:GetChildren()) do
if isCosmetic(o) then
local ok, c = pcall(function() return o:Clone() end)
if ok and c then c.Parent = m end
end
end
local head = ch:FindFirstChild("Head")
if head then
local hp = Instance.new("Part")
hp.Name = "Head"
hp.Color = head.Color
hp.Parent = m
local f = head:FindFirstChildOfClass("Decal")
if f then f:Clone().Parent = hp end
local sm = head:FindFirstChildOfClass("SpecialMesh")
if sm then sm:Clone().Parent = hp end
end
return m
end

local hum = getHum()

if not id then
local ok, res = pcall(function() return Players:GetUserIdFromNameAsync(q) end)
if ok then id = res end
end
if not id then error("pemain '" .. q .. "' tidak ditemukan") end
if id == player.UserId and not original then error("itu akun kamu sendiri") end
if not original then original = snapshot() end
if p and p ~= player and p.Character and p.Character:FindFirstChildOfClass("Humanoid") then
local ok, sum, face = pcall(applyModel, p.Character)
if ok then
if not face then
pcall(function()
local m = Players:GetCharacterAppearanceAsync(id)
if m then
if copyFace(m) then sum = sum:gsub("wajah %-", "wajah ok(akun)") end
m:Destroy()
end
end)
end
return "Disalin dari " .. p.Name .. ": " .. sum
end
warn("[JxL] copy cara 1 gagal: " .. tostring(sum))
end
local ok2, model = pcall(function() return Players:GetCharacterAppearanceAsync(id) end)
if ok2 and model then
local ok, sum = pcall(applyModel, model)
pcall(function() model:Destroy() end)
if ok then return "Disalin (ID " .. id .. "): " .. sum end
warn("[JxL] copy cara 2 gagal: " .. tostring(sum))
else
warn("[JxL] GetCharacterAppearanceAsync gagal: " .. tostring(model))
end
local desc = Players:GetHumanoidDescriptionFromUserId(id)
hum:ApplyDescription(desc)
return "Disalin lewat HumanoidDescription (ID " .. id .. ")"
end

local function attachAcc(hum, ch, id, layered)
local ok, objs = pcall(function() return game:GetObjects("rbxassetid://" .. id) end)
if not ok or type(objs) ~= "table" then return false, "GetObjects diblokir" end
local acc
for _, o in ipairs(objs) do
if o:IsA("Accessory") then acc = o break end
local d = o:FindFirstChildWhichIsA("Accessory", true)
if d then acc = d break end
end
if not acc then return false, "item bukan aksesori" end
if layered then
pcall(function() acc:Destroy() end)
return false, "item berlapis hanya bisa dipasang lewat avatar R15"
end
acc:SetAttribute("JxLCat", id)
local h = acc:FindFirstChild("Handle")
if h then
for _, w in ipairs(h:GetChildren()) do
if w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D") then w:Destroy() end
end
end
pcall(function() hum:AddAccessory(acc) end)
if not acc.Parent then acc.Parent = ch end
task.wait()
local function welded()
if not h then return false end
for _, w in ipairs(h:GetChildren()) do
if (w:IsA("Weld") or w:IsA("Motor6D")) and w.Part1 and w.Part1:IsDescendantOf(ch) then return true end
end
return false
end
if welded() then return true end
local att = h and h:FindFirstChildOfClass("Attachment")
if not att then pcall(function() acc:Destroy() end) return false, "aksesori tanpa attachment" end
for _, d in ipairs(ch:GetDescendants()) do
if d:IsA("Attachment") and d.Name == att.Name and d.Parent ~= h and not d:IsDescendantOf(acc) then
local weld = Instance.new("Weld")
weld.Name = "AccessoryWeld"
weld.Part0, weld.Part1 = h, d.Parent
weld.C0, weld.C1 = att.CFrame, d.CFrame
weld.Parent = h
return true
end
end
pcall(function() acc:Destroy() end)
return false, "titik tempel tidak ditemukan"
end

local function buildDesc()
local desc = original:Clone()
for _, w in ipairs(worn) do
if w.slot then desc[w.slot] = w.id end
end
local list = {}
pcall(function() list = original:GetAccessories(true) end)
local drop = {}
local function dropTypes(names) for _, n in ipairs(names) do drop[n] = true end end
for _, w in ipairs(worn) do
if w.acc and w.acc ~= "Hat" then drop[w.acc] = true end
if w.slot == "Shirt" then dropTypes({"TShirt", "Shirt", "Jacket", "Sweater"}) end
if w.slot == "GraphicTShirt" then dropTypes({"TShirt"}) end
if w.slot == "Pants" then dropTypes({"Pants", "Shorts", "DressSkirt"}) end
end
local kept, have = {}, {}
for _, a in ipairs(list) do
local tn
pcall(function() tn = a.AccessoryType.Name end)
if tn and drop[tn] then
-- diganti item baru
else
table.insert(kept, a)
have[a.AssetId] = true
end
end
for _, w in ipairs(worn) do
if w.acc and not have[w.id] then
local okT, at = pcall(function() return Enum.AccessoryType[w.acc] end)
if okT and at then
table.insert(kept, {AssetId = w.id, AccessoryType = at, IsLayered = w.layered and true or false, Order = 1, Puffiness = 1})
have[w.id] = true
end
end
end
-- susun ulang urutan lapisan: kaus -> baju -> celana -> rok -> sweater -> jaket -> sepatu
local lay = {}
for i, a in ipairs(kept) do
if a.IsLayered then
local tn = "?"
pcall(function() tn = a.AccessoryType.Name end)
table.insert(lay, {rank = layeredOrder[tn] or 1, idx = i, e = a})
end
end
table.sort(lay, function(x, y)
if x.rank ~= y.rank then return x.rank < y.rank end
return x.idx < y.idx
end)
for i, r in ipairs(lay) do
r.e.Order = i
if not r.e.Puffiness then r.e.Puffiness = 1 end
end
desc:SetAccessories(kept, true)
return desc
end
local classicSlots = {Shirt = true, Pants = true, GraphicTShirt = true}
local function applyAll()
applyNote = nil
local hum, ch = getHum()
if not hum then return false, "karakter belum siap" end
if not original then
local ok, d = pcall(function() return hum:GetAppliedDescription() end)
if not ok or not d then return false, "tampilan awal tidak terbaca" end
original = d
end

for _, w in ipairs(worn) do
if w.layered and isR6 then return false, "item berlapis butuh avatar R15 (avatarmu R6)" end
end
local anySlot = false
for _, w in ipairs(worn) do if w.slot then anySlot = true end end
local okD, desc = pcall(buildDesc)
local viaDesc = false
if okD and desc then
viaDesc = pcall(function() hum:ApplyDescription(desc) end)
end
if viaDesc then
hadSlots = true
-- aksesori hasil fallback lama yang tidak dipakai lagi dibersihkan

for _, w in ipairs(worn) do if w.acc then keep[w.id] = true end end
for _, c in ipairs(ch:GetChildren()) do
local tag = c:IsA("Accessory") and c:GetAttribute("JxLCat")

end

local okV, applied = pcall(function() return hum:GetAppliedDescription() end)

if okV and applied then
pcall(function()
for _, a in ipairs(applied:GetAccessories(true)) do appliedAcc[a.AssetId] = true end

end
local missing
for _, w in ipairs(worn) do

missing = w
elseif w.slot and classicSlots[w.slot] and okV and applied then
local okS, cur = pcall(function() return applied[w.slot] end)
if okS and cur ~= w.id then applyNote = "baju/celana klasik mungkin ditolak game" end
end
end
if not missing then return true end
if missing.layered or missing.slot then
return false, "item ditolak game (tidak muncul di avatar)"
end
-- aksesori biasa yang tidak muncul: coba cadangan manual
for _, w in ipairs(worn) do
if w.acc and not appliedAcc[w.id] and not findTagged(ch, w.id) then
local ok, why = attachAcc(hum, ch, w.id, w.layered)
if not ok then return false, why end
end
end
return true
end
-- ApplyDescription gagal total: cadangan manual (hanya aksesori biasa)
local failWhy
local keep = {}
for _, w in ipairs(worn) do if w.acc then keep[w.id] = true end end
for _, c in ipairs(ch:GetChildren()) do
local tag = c:IsA("Accessory") and c:GetAttribute("JxLCat")
if tag and not keep[tag] then c:Destroy() end
end
if anySlot then failWhy = "game memblokir ApplyDescription (baju/badan tidak bisa diganti)" end
for _, w in ipairs(worn) do
if w.acc and not findTagged(ch, w.id) then
local ok, why = attachAcc(hum, ch, w.id, w.layered)
if not ok then failWhy = why end
end
end
if failWhy then return false, failWhy end
return true
end
local function infoOf(id)
local ok, info = pcall(function() return Market:GetProductInfo(id, Enum.InfoType.Asset) end)
if ok and info then return info end
end
local function makeEntry(id)
local info = infoOf(id)
if not info then return nil, "ID tidak ditemukan" end
local t = info.AssetTypeId
if slotMap[t] then return {id = id, slot = slotMap[t]} end
if accMap[t] then return {id = id, acc = accMap[t], layered = (t >= 64 and t <= 72)} end
return nil, "jenis item tidak didukung"
end
local httpReq = (syn and syn.request) or http_request or request or (http and http.request)
local function bundleAssets(bid)
if not httpReq then return nil end
local ok, res = pcall(httpReq, {Url = "https://catalog.roblox.com/v1/bundles/" .. bid .. "/details", Method = "GET"})
if not ok or not res or not res.Body then return nil end
local ok2, d = pcall(function() return HttpS:JSONDecode(res.Body) end)
if not ok2 or type(d) ~= "table" or type(d.items) ~= "table" then return nil end
local out = {}
for _, it in ipairs(d.items) do
if it.type == "Asset" and it.id then table.insert(out, it.id) end
end
return out
end
local function removeId(id)
for i = #worn, 1, -1 do if worn[i].id == id then table.remove(worn, i) end end
end
local function evictConflicts(e)
for i = #worn, 1, -1 do
local w = worn[i]
if w.id ~= e.id then
if (e.slot and w.slot == e.slot) or (e.acc and e.acc ~= "Hat" and w.acc == e.acc) then
table.remove(worn, i)
end
end
end
end
local function setWorn(id, bundle, want)
if busy then say("Tunggu sebentar...") return end
busy = true
task.spawn(function()
local okRun, errRun = pcall(function()
local hum = getHum()
if not hum then say("Karakter belum siap") return end
if #worn == 0 and want then original = nil end -- tampilan asli diambil hanya saat mulai memakai, bukan saat melepas
if bundle then
if want then
local ids = bundleAssets(id)
if not ids or #ids == 0 then say("Isi bundle tidak terbaca") return end
local n = 0
for _, aid in ipairs(ids) do
local e = makeEntry(aid)
if e then e.bundle = id removeId(aid) evictConflicts(e) table.insert(worn, e) n = n + 1 end
end
if n == 0 then say("Bundle tidak didukung") return end
bundleWorn[id] = true
else
for i = #worn, 1, -1 do if worn[i].bundle == id then table.remove(worn, i) end end
bundleWorn[id] = nil
end
else
if want then
local e, why = makeEntry(id)
if not e then say("Gagal: " .. tostring(why)) return end
removeId(id)
evictConflicts(e)
table.insert(worn, e)
else
removeId(id)
end
end
local ok, err = applyAll()
if not ok and want then
if bundle then
for i = #worn, 1, -1 do if worn[i].bundle == id then table.remove(worn, i) end end
bundleWorn[id] = nil
else
removeId(id)
end
pcall(applyAll)
end
say(ok and ((want and "Terpasang" or "Dilepas") .. (applyNote and (" - " .. applyNote) or "")) or ("Gagal: " .. tostring(err)))
end)
if not okRun then say("Error: " .. tostring(errRun)) end
busy = false
refreshRows()
end)
end

local function search(q)
local c = cats[cat][2]
local res = {}
local sm = sorts[sortI][2]
local ok = pcall(function()
local AES = game:GetService("AvatarEditorService")
local p = CatalogSearchParams.new()
p.SearchKeyword = q
p.IncludeOffSale = (sm == "ltd")
local function setEnum(prop, enumName, names)
for _, n in ipairs(names) do
local ok2, v = pcall(function() return Enum[enumName][n] end)
if ok2 and v then pcall(function() p[prop] = v end) return end
end
end
if sm == "fav" then setEnum("SortType", "CatalogSortType", {"MostFavorited"})
elseif sm == "new" then setEnum("SortType", "CatalogSortType", {"RecentlyCreated"})
else setEnum("SortType", "CatalogSortType", {"Bestselling", "BestSelling"}) end
setEnum("SortAggregation", "CatalogSortAggregation", {sm == "sell" and "PastMonth" or "AllTime"})
if sm == "ltd" then setEnum("CategoryFilter", "CatalogCategoryFilter", {"Collectibles"}) end
if c == "bundle" then
p.BundleTypes = {Enum.BundleType.BodyParts}
elseif c then
p.AssetTypes = resolveEnums(c)
end
local pages = AES:SearchCatalog(p)
for _, it in ipairs(pages:GetCurrentPage()) do
local isB = tostring(it.ItemType):find("Bundle", 1, true) ~= nil
table.insert(res, {id = it.Id, name = it.Name or "?", bundle = isB})
end
end)
if ok and #res > 0 then return res end
if httpReq then
pcall(function()
local sortNum = ({sell = 2, fav = 1, ltd = 2, new = 3})[sm]
local url = "https://catalog.roblox.com/v1/search/items/details?Keyword=" .. HttpS:UrlEncode(q) .. "&Limit=30&SortType=" .. sortNum
if sm == "ltd" then url = url .. "&SalesTypeFilter=2&IncludeNotForSale=true" end
local r = httpReq({Url = url, Method = "GET"})
local d = HttpS:JSONDecode(r.Body)
for _, it in ipairs(d.data or {}) do
table.insert(res, {id = it.id, name = it.name or "?", bundle = it.itemType == "Bundle"})
end
end)
end
return res
end
